\newcommand{\tracecurve}[6]{%
  \begin{scope}[shift={(#1,#2)},xscale=#3,yscale=#4]
    \draw[#5,line width=.85pt,line join=round] plot coordinates {#6};
  \end{scope}}
\newcommand{\smallplot}[6]{%
  \draw[draw=#5!75,fill=white,line width=.75pt] (#1,#2) rectangle ++(#3,#4);
  \tracecurve{#1+.10}{#2+#4-.14}{#3-.20}{#4-.32}{#5}{#6}}
% Compact row of spectral tokens, with an ellipsis near the right edge.
\newcommand{\tokens}[4]{%
  \begin{scope}[shift={(#1,#2)}]
    \foreach \k in {0,1,2,3}{
      \draw[draw=#4!85!black,fill=#4!12,line width=.7pt]
        ({\k*#3/6.5},0) rectangle ++({#3/6.5},.43);}
    \node[inner sep=0] at ({4.65*#3/6.5},.18) {$\cdots$};
    \draw[draw=#4!85!black,fill=#4!12,line width=.7pt]
      ({5.5*#3/6.5},0) rectangle ++({#3/6.5},.43);
  \end{scope}}
\newcommand{\contexttokens}[3]{%
  \foreach \k in {0,...,4}{\draw[flow!85!black,fill=flow!15,line width=.6pt]
    ({#1+\k*#3/5},#2) rectangle ++({#3/5},.30);}}
% Stacked heatmaps preserve the reference motif; the cell fills are schematic.
\newcommand{\heatmap}[4]{%
  \begin{scope}[shift={(#1,#2)}]
    \foreach \z in {3,2,1}{\draw[draw=#3,line width=.65pt,fill=white]
      ({.12*\z},{-.12*\z}) rectangle ++(2.8,2.08);}
    \fill[white] (0,0) rectangle (2.8,2.08);
    \csname #3cells\endcsname
    \draw[#3,line width=.85pt] (0,0) rectangle (2.8,2.08);
    \ifnum#4=1
      \foreach \j in {0,...,39}{\pgfmathtruncatemacro{\shade}{100-\j*2.5}
        \fill[#3!\shade] (3.44,{-.23+\j*.055}) rectangle ++(.14,.058);}
      \node[tinylabel,anchor=west] at (3.64,-.20) {1};
      \node[tinylabel,anchor=west] at (3.64,1.91) {0};
    \fi
    \node[small,rotate=90] at (-.32,1.02) {Map row};
  \end{scope}}
